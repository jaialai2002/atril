# fuera de los buscadores: ningún archivo del Atril (páginas, audio, partituras, json) se indexa
<IfModule mod_headers.c>
  Header always set X-Robots-Tag "noindex, nofollow, noarchive, nosnippet, noimageindex"
</IfModule>

# Atril · atril.sigan.me
DirectoryIndex index.html
Options -Indexes

# la lista de obras es privada: solo la entrega lista.php a quien entra con Google
<Files "obras.json">
  <IfModule mod_authz_core.c>
    Require all denied
  </IfModule>
  <IfModule !mod_authz_core.c>
    Order allow,deny
    Deny from all
  </IfModule>
</Files>

# tipos de archivo (audio de las pistas y partituras)
AddType audio/mp4 .m4a
AddType application/vnd.recordare.musicxml+xml .musicxml
AddType application/json .json
AddCharset utf-8 .html .json .musicxml

# comprimir lo que es texto (la partitura baja ~6 veces más ligera)
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html application/json application/vnd.recordare.musicxml+xml
</IfModule>

# el índice, la página y los obra.json siempre al día (sin quedarse con una copia vieja)
<IfModule mod_headers.c>
  <FilesMatch "^(index\.html|obras\.json|obra\.json)$">
    Header set Cache-Control "no-cache"
  </FilesMatch>
</IfModule>

# las pistas, la partitura y las fuentes llevan ?v=<fecha de la versión> (cambia con cada subida desde el Taller): el
# navegador las guarda un año sin volver a preguntar al servidor. Al reabrir una obra (o la siguiente de un concierto,
# que el Atril precarga) el audio sale de la caché del iPad en vez de bajar otra vez 8 o 9 pistas.
<IfModule mod_headers.c>
  <FilesMatch "\.(m4a|musicxml|woff2)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
</IfModule>

# enlace limpio de cada obra: atril.sigan.me/pastorella-in-g lo sirve obra.php (con la tarjeta de la obra para WhatsApp)
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteRule ^([a-z0-9-]+)/?$ obra.php?id=$1 [L]
</IfModule>
